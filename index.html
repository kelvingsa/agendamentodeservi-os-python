'border-slate-700/60 shadow-none' : 'border-slate-100 shadow-slate-200/40'}; document.getElementById('action-bar').className = ${cfg.cardClass} mt-6 p-4 rounded-xl border shadow-md flex flex-wrap justify-between items-center gap-4 transition-all duration-500 ${isNoite ? 'border-slate-700/60' : 'border-slate-100'}; // Ajuste de cores de realce document.getElementById('brand-logo').style.backgroundColor = cfg.primColor; document.getElementById('form-title').style.color = cfg.primColor; document.getElementById('table-title').style.color = cfg.primColor; document.getElementById('btn-submit').style.backgroundColor = cfg.primColor; document.getElementById('table-header').style.backgroundColor = cfg.primColor; // Caixa de informações secundária document.getElementById('info-box').className = mt-8 p-4 rounded-xl flex gap-3 text-xs leading-relaxed transition-all duration-500 ${isNoite ? Que tal escolher outro momento perfeito para o cliente?); return; } agendamentos.push({ id: proximoId++, cliente, servico, profissional, data, horario }); document.getElementById('form-agendamento').reset(); document.getElementById('data').value = new Date().toISOString().split('T')[0]; atualizarTabela(agendamentos); alert(" Excelente! O horário de atendimento foi reservado e salvo com sucesso!"); } function liberarHorario() { if (!idSelecionado) { alert(" Dica: Clique primeiro em cima da linha do cliente desejado na tabela para poder liberar o horário."); return; } agendamentos = agendamentos.filter(ag => ag.id !== idSelecionado); idSelecionado = null; atualizarTabela(agendamentos); alert(" Agenda atualizada!
